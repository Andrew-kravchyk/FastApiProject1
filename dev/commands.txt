# 1. Зупиняє та видаляє контейнери, мережі і томи (volumes)
docker compose down -v

# 2. Видаляє всі невикористані контейнери, образи, мережі та кеш
docker system prune -a -f

# 3. Очищає кеш збірки образів
docker builder prune -a -f

# 4. Видаляє всі невикористані томи
docker volume prune -f

# 5. Збирає образи заново без використання кешу
docker compose build --no-cache

# 6. Запускає контейнери у фоновому режимі
docker compose up -d

# 7. Показує статус запущених контейнерів
docker compose ps

# 8. Показує логи контейнера api (для перевірки autoreload)
docker compose logs -f api

# 9. Показує список встановлених Python-пакетів у контейнері
docker compose exec api pip list

# 10. Показує залежності через Poetry у контейнері
docker compose exec api poetry show

# 11. Зупиняє всі сервіси проекту
docker compose down
